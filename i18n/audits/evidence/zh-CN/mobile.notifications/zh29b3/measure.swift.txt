func font(_ size: CGFloat, _ weight: NSFont.Weight) -> NSFont {
  if let base = NSFont(name: "PingFang SC", size: size) {
    let traits: NSFontTraitMask = weight >= .semibold ? .boldFontMask : []
    return NSFontManager.shared.convert(base, toHaveTrait: traits)
  }
  return NSFont.systemFont(ofSize: size, weight: weight)
}
func width(_ c: Case, _ scale: CGFloat) -> CGFloat {
  let size = c.size * scale
  let f = font(size, c.weight)
  let w = (c.text as NSString).size(withAttributes: [.font: f]).width
  return w + c.tracking * size * CGFloat(max(0, c.text.count - 1))
}

  let cells = c.text.unicodeScalars.reduce(0) { n, u in
    let v = u.value
    let wide = (v >= 0x1100 && v <= 0x115f) || (v >= 0x2e80 && v <= 0xa4cf) || (v >= 0xac00 && v <= 0xd7a3) || (v >= 0xf900 && v <= 0xfaff) || (v >= 0xfe10 && v <= 0xfe19) || (v >= 0xfe30 && v <= 0xfe6f) || (v >= 0xff00 && v <= 0xff60) || (v >= 0xffe0 && v <= 0xffe6)
    return n + (wide ? 2 : 1)
  }
  print(String(format:"%@\t%@\t%d\t%d\t%.1f\t%.0f\t%d\t%.1f\t%d\t%@", c.group, c.label, chars, cells, w1, c.avail, Int(ceil(w1/c.avail)), w2, Int(ceil(w2/c.avail)), c.note))
}
