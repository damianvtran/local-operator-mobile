type Node = any;
const CLDR = new Intl.PluralRules("zh-CN").resolvedOptions().pluralCategories;

function collectText(nodes: Node[], into: string[] = []): string[] {
	for (const n of nodes) {
		if (n.type === "text") into.push(n.value);
		else if (n.type === "plural" || n.type === "select") {
			for (const [, arms] of n.options as Map<string, Node[]>) collectText(arms, into);
		}
	}
	return into;
}
function selectSignatures(nodes: Node[], into: string[][] = []): string[][] {
	for (const n of nodes) {
		if (n.type === "select") {
			into.push([...n.options.keys()]);
			for (const [, arms] of n.options as Map<string, Node[]>) selectSignatures(arms, into);
		} else if (n.type === "plural") {
			for (const [, arms] of n.options as Map<string, Node[]>) selectSignatures(arms, into);
		}
	}
	return into;
}
function pluralArms(nodes: Node[], into: string[][] = []): string[][] {
	for (const n of nodes) {
		if (n.type === "plural") {
			into.push([...n.options.keys()]);
			for (const [, arms] of n.options as Map<string, Node[]>) pluralArms(arms, into);
		} else if (n.type === "select") {
			for (const [, arms] of n.options as Map<string, Node[]>) pluralArms(arms, into);
		}
	}
	return into;
}
const chars = (s: string): number => [...s].length;
const isWide = (ch: string): boolean => {
	const cp = ch.codePointAt(0)!;
	return (
		(cp >= 0x1100 && cp <= 0x115f) ||
		(cp >= 0x2e80 && cp <= 0xa4cf) ||
		(cp >= 0xac00 && cp <= 0xd7a3) ||
		(cp >= 0xf900 && cp <= 0xfaff) ||
		(cp >= 0xfe10 && cp <= 0xfe19) ||
		(cp >= 0xfe30 && cp <= 0xfe6f) ||
		(cp >= 0xff00 && cp <= 0xff60) ||
		(cp >= 0xffe0 && cp <= 0xffe6)
	);
};
const cells = (s: string): number => [...s].reduce((n, ch) => n + (isWide(ch) ? 2 : 1), 0);

let mechFail = 0;
for (const key of Object.keys(EN)) {
	const en = parseMessage(EN[key]);
	const zh = parseMessage(ZH[key]);
	const problems: string[] = [];
	const enArgs = [...en.params.entries()].map(([n, k]) => `${n}:${k}`);
	const zhArgs = [...zh.params.entries()].map(([n, k]) => `${n}:${k}`);
	if (enArgs.join("|") !== zhArgs.join("|")) problems.push(`args en=[${enArgs}] zh=[${zhArgs}]`);
	const enSel = JSON.stringify(selectSignatures(en.nodes));
	const zhSel = JSON.stringify(selectSignatures(zh.nodes));
	if (enSel !== zhSel) problems.push(`selects en=${enSel} zh=${zhSel}`);
	const zhPlural = pluralArms(zh.nodes);
	for (const arms of zhPlural) {
		const extra = arms.filter((a) => !CLDR.includes(a) && !a.startsWith("="));
		const missing = CLDR.filter((a) => !arms.includes(a));
		if (extra.length || missing.length) problems.push(`plural arms=${JSON.stringify(arms)} extra=${JSON.stringify(extra)} missing=${JSON.stringify(missing)}`);
	}
	const text = collectText(zh.nodes).join("");
	const latin = text.replace(/lop mobile|Radient/g, "").match(/[A-Za-z]+/g);
	if (latin) problems.push(`latin text=${latin.join(",")}`);
	if (/[\u202A-\u202E\u2066-\u2069]/.test(ZH[key])) problems.push("bidi control");
	if (/[,.;:!?()]/.test(text)) problems.push("ascii prose punctuation");
	const status = problems.length === 0 ? "PASS" : "FAIL";
	if (problems.length) mechFail += 1;
	console.log(
		`MECH ${key}: ${status}; args=${zhArgs.join(",") || "none"}; selects=${selectSignatures(zh.nodes).length}; plural=${JSON.stringify(zhPlural)}; chars_en=${chars(EN[key])}; chars_zh=${chars(ZH[key])}; cells_zh=${cells(ZH[key])}${problems.length ? `; ${problems.join("; ")}` : ""}`,
	);
}

const dueLabel = (seconds: number): string => {
	const magnitude = Math.abs(seconds);
	let label: string;
	if (magnitude < 90) label = `${Math.floor(magnitude)}s`;
	else if (magnitude < 5400) label = `${Math.floor(magnitude / 60)}m`;
	else if (magnitude < 172800) label = `${Math.floor(magnitude / 3600)}h`;
	else label = `${Math.floor(magnitude / 86400)}d`;
	return seconds < 0 ? render("mobile.schedules.due.overdue", { label }) : render("mobile.schedules.due.ahead", { label });
};
const dueLabelEn = (seconds: number): string => {
	const magnitude = Math.abs(seconds);
	let label: string;

const nextDueClause = (head: string, seconds: number): string => {

	return seconds < 0

};
