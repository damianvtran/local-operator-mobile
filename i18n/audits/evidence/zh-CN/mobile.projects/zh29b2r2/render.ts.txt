/* ---------------------------------------------------------------- resolver */

const HAS_EXT = /\.[cm]?[jt]s$|\.json$/;
registerHooks({
	resolve(specifier, context, nextResolve) {
		if (specifier.startsWith("@/")) {
			const base = `${SRC}/${specifier.slice(2)}`;
			for (const candidate of [`${base}.ts`, `${base}/index.ts`, `${base}.json`]) {
				if (existsSync(candidate) && statSync(candidate).isFile()) {
					return nextResolve(pathToFileURL(candidate).href, context);
				}
			}
		}
		if (
			(specifier.startsWith("./") || specifier.startsWith("../")) &&
			context.parentURL?.startsWith("file:") &&
			!HAS_EXT.test(specifier)
		) {
			const base = fileURLToPath(new URL(specifier, context.parentURL).href);
			for (const candidate of [`${base}.ts`, `${base}/index.ts`]) {
				if (existsSync(candidate) && statSync(candidate).isFile()) {
					return nextResolve(pathToFileURL(candidate).href, context);
				}
			}
		}
		return nextResolve(specifier, context);
	},
});

type Node = any;
const walk = (nodes: readonly Node[], visit: (n: Node) => void): void => {
	for (const n of nodes) {
		visit(n);
		if (n.type === "plural" || n.type === "select") {
			for (const [, arms] of n.options as Map<string, readonly Node[]>) {
				walk(arms, visit);
			}
		}
	}
};
const plurals = (nodes: readonly Node[]): string[][] => {
	const out: string[][] = [];
	walk(nodes, (n) => {
		if (n.type === "plural") out.push([...n.options.keys()]);
	});
	return out;
};
const selects = (nodes: readonly Node[]): string[][] => {
	const out: string[][] = [];
	walk(nodes, (n) => {
		if (n.type === "select") out.push([...n.options.keys()]);
	});
	return out;
};
const chars = (s: string): number => [...s].length;
const isWide = (ch: string): boolean => {
	const cp = ch.codePointAt(0) ?? 0;
	return (
		(cp >= 0x1100 && cp <= 0x115f) ||
		(cp >= 0x2e80 && cp <= 0xa4cf) ||
		(cp >= 0xac00 && cp <= 0xd7a3) ||
		(cp >= 0xf900 && cp <= 0xfaff) ||
		(cp >= 0xfe10 && cp <= 0xfe19) ||
		(cp >= 0xfe30 && cp <= 0xfe6f) ||
		(cp >= 0xff00 && cp <= 0xff60) ||
		(cp >= 0xffe0 && cp <= 0xffe6)
	);
};
const cells = (s: string): number =>
	[...s].reduce((n, ch) => n + (isWide(ch) ? 2 : 1), 0);

const ASCII_PROSE = /[,.?!;:]/;

	let pe: any;
	let pz: any;
	try {
		pe = parseMessage(en[key]);
		pz = parseMessage(zh[key]);
	} catch (error) {
		parseFail += 1;
		line(`PARSE FAIL ${key}: ${(error as Error).message}`);
		continue;
	}
	const shape = (p: any) =>
		JSON.stringify([...p.params.entries()].sort((a: any, b: any) => (a[0] < b[0] ? -1 : 1)));
	if (shape(pe) !== shape(pz)) {
		paramFail += 1;
		line(`PARAM MISMATCH ${key}\n  en=${shape(pe)}\n  zh=${shape(pz)}`);
	}
	const enP = plurals(pe.nodes);
	const zhP = plurals(pz.nodes);
	if (plurals(pe.nodes).length !== plurals(pz.nodes).length) {
		paramFail += 1;
		line(`PLURAL COUNT MISMATCH ${key}: en=${enP.length} zh=${zhP.length}`);
	}
	const enS = selects(pe.nodes);
	const zhS = selects(pz.nodes);
	if (enS.length !== zhS.length) {
		paramFail += 1;
		line(`SELECT COUNT MISMATCH ${key}: en=${enS.length} zh=${zhS.length}`);
	} else {
		enS.forEach((arms, i) => {
			const mine = zhS[i];
			const enSet = arms.filter((a) => a !== "other");
			const zhSet = mine.filter((a) => a !== "other");

				paramFail += 1;
				line(`SELECT ARM MISMATCH ${key} #${i}: en=${JSON.stringify(arms)} zh=${JSON.stringify(mine)}`);
			}
			if (!mine.includes("other")) {
				paramFail += 1;
				line(`SELECT MISSING other ${key} #${i}: ${JSON.stringify(mine)}`);
			}
		});
	}
}

rule("MECHANICAL — bidi controls / ASCII prose punctuation / Latin residue");

		if (/^(other|wakes|monitors)$/.test(run.trim())) continue;
		latinHits.push(`${key}: ${JSON.stringify(run.trim())}`);
	}
}

/** Run the app's own composition functions with an English preference, then put
 * the reader back on zh-CN. `t()` inside `dueLabel`/`nextDueClause` resolves the
 * locale source at call time, so this is the same path an English reader takes,
 * not a hand-built comparator string. */
const withLocale = <T,>(locale: string, fn: () => T): T => {
	setLocaleSource(() => locale);
	try {
		return fn();
	} finally {
		setLocaleSource(() => "zh-CN");
	}
};
