import { readFileSync } from "node:fs";
import { createHash } from "node:crypto";

const BASE = "/Users/damian/workspace/mobile-i18n-d5aaf8ba6ba5/mobile-i18n";

const json = (p) => JSON.parse(readFileSync(`${BASE}/${p}`, "utf8"));
const sha = (p) => createHash("sha256").update(readFileSync(`${BASE}/${p}`)).digest("hex");

const EN_HOME = json("src/i18n/catalogues/en/mobile.home.json");
const CTX_HOME = json("src/i18n/catalogues/en/mobile.home.context.json");

console.log(H("input bytes audited (sha256, current working tree)"));
for (const p of [
	"src/i18n/catalogues/en/mobile.home.json",
	"src/i18n/catalogues/en/mobile.home.context.json",
	"src/i18n/catalogues/fr/mobile.home.json",

console.log(`fr label: ${lab(FR_HOME[K1], FR_SESSION["mobile.session.conversation_details"])}`);

{
	const v = FR_HOME[K1];
	const codes = [...v].map((c) => c.codePointAt(0));
	const nonAscii = codes.filter((c) => c > 0x7f).map((c) => `U+${c.toString(16).toUpperCase().padStart(4, "0")}`);

	console.log(`bidi/control chars: ${codes.filter((c) => (c >= 0x200b && c <= 0x200f) || (c >= 0x202a && c <= 0x202e) || (c >= 0x2066 && c <= 0x2069) || c === 0x061c || c === 0x00a0).map((c) => `U+${c.toString(16).toUpperCase()}`).join(" ") || "none"}`);

console.log(H("5. mobile.projects.write_receipt — source, context, target"));

console.log(H("5a. declared arms per locale (from the app's own parser)"));
const arms = (src) => {
	const out = [];
	for (const node of parseMessage(src).nodes) {
		if (node.type === "select" || node.type === "plural") out.push([node.type, node.name, [...node.options.keys()].sort()]);
	}
	return out;
};
console.log(`en: ${JSON.stringify(arms(EN_PROJ[K2]))}`);
console.log(`fr: ${JSON.stringify(arms(FR_PROJ[K2]))}`);
console.log(`params: en=${JSON.stringify([...parseMessage(EN_PROJ[K2]).params])} fr=${JSON.stringify([...parseMessage(FR_PROJ[K2]).params])}`);

console.log(H("6. all seven codes + other, rendered through renderFor (en / fr)"));
const CODES = ["created", "completed", "reopened", "added", "updated", "removed", "deleted", "other"];
const ENGLISH_VERBS = /Created|Completed|Reopened|Added|Updated|Removed|Deleted/;
for (const code of CODES) {
	const en = renderFor(K2, { action: code, subject: "payments-migration" }, "en");
	const fr = renderFor(K2, { action: code, subject: "Projet X" }, "fr");
	const leak = ENGLISH_VERBS.test(fr);
	console.log(`${code.padEnd(9)} en=${JSON.stringify(en).padEnd(32)} fr=${JSON.stringify(fr).padEnd(26)} english-verb-in-fr=${leak}`);
}
