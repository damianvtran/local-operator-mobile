def sha(data):
    return hashlib.sha256(data).hexdigest()


def no_duplicate_keys(pairs):
    seen = {}
    for key, value in pairs:
        if key in seen:
            raise ValueError(f"duplicate key {key!r}")
        seen[key] = value
    return seen


def load(path):
    raw = open(f"{REPO}/{path}", "rb").read()
    try:
        data = json.loads(raw.decode("utf-8"), object_pairs_hook=no_duplicate_keys)
    except ValueError as exc:
        print(f"{path}: INVALID JSON or duplicate key: {exc!r}")
        sys.exit(1)
    print(f"{path}: valid JSON, no duplicate keys, {len(data)} keys, sha256 {sha(raw)}")
    return data

def kind_arms(value):
    """Arms of the ROOT select, by brace depth. The regex-free scan matters: the
    arm label `milestone` also appears inside the en milestone body as a noun,
    so a substring count would over-report it."""
    index = value.index("select,") + len("select,")
    depth = 1
    arms = {}
    i = index
    while i < len(value):
        ch = value[i]
        if ch == "{":
            depth += 1
            i += 1
            continue
        if ch == "}":
            depth -= 1
            i += 1
            if depth == 0:
                break
            continue
        if depth == 1:
            match = re.match(r"\s*([A-Za-z][A-Za-z0-9_]*)\s*", value[i:])
            if match:
                arm = match.group(1)
                j = i + match.end()
                assert value[j] == "{", (arm, value[j:j + 10])
                inner_depth = 0
                k = j
                while k < len(value):
                    if value[k] == "{":
                        inner_depth += 1
                    elif value[k] == "}":
                        inner_depth -= 1
                        if inner_depth == 0:
                            break
                    k += 1
                arms[arm] = value[j + 1:k]
                i = k + 1
                continue
        i += 1
    return arms


PAIR = re.compile(r"(created|completed|reopened|added|updated|removed|deleted) \{([^}]*)\}")
for label, value in (("en", en[KEY]), ("fr", fr[KEY])):
    arms = kind_arms(value)
    print(f"{label}: kind arms {sorted(arms)}; each arm body:")
    for arm in sorted(arms):
        body = arms[arm]
        codes = [c for c, _ in PAIR.findall(body)]
        print(f"  {label}.{arm:9s} action arms {len(codes)} {codes} ; body={body!r}")
    print(
        f"{label}: {{subject}} x{value.count('{subject}')}; {{kind, select x{value.count('{kind, select')}; "
        f"{{action, select x{value.count('{action, select')}; "
        f"starts {{kind, select: {value.startswith('{kind, select,')}; ends '.}}': {value.endswith('.}}')}"
    )

print("\n=== D. action arms per locale, byte for byte against the expected words ===")
for label, value, table in (("en", en[KEY], EN_WORDS), ("fr", fr[KEY], FR_WORDS)):
    per_code = {}
    for code, word in PAIR.findall(value):
        per_code.setdefault(code, set()).add(word)
    ok = all(per_code.get(code) == {table[code]} for code in CODES)
    print(f"{label}: every code maps to exactly one expected word: {ok}")
    for code in CODES:
        print(f"  {label} {code:9s} -> {sorted(per_code.get(code, []))}")
