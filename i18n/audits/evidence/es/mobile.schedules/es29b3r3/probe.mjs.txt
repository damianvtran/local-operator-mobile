const kindOf = (m) => [...m.params.entries()].map(([n, k]) => `${n}:${k}`).sort().join(",");
const armsOf = (nodes, name) => {
	const found = [];
	const walk = (ns) => {
		for (const n of ns) {
			if (n.type === "plural" && (name === null || n.name === name)) {
				found.push({ name: n.name, arms: [...n.options.keys()] });
			}
			if (n.type === "plural" || n.type === "select") {
				for (const [, v] of n.options) walk(v);
			}
		}
	};
	walk(nodes);
	return found;
};

H("2. ICU PARSE (es) + PLACEHOLDER NAME/TYPE PARITY vs en");
let parseFail = 0, paramMismatch = 0;
for (const k of enKeys) {
	let pe, ps;
	try { pe = M.parseMessage(en[k]); } catch (e) { say(`PARSE-FAIL en ${k}: ${e.message}`); parseFail++; continue; }
	try { ps = M.parseMessage(es[k]); } catch (e) { say(`PARSE-FAIL es ${k}: ${e.message}`); parseFail++; continue; }
	const a = kindOf(pe), b = kindOf(ps);
	if (a !== b) { say(`KIND-DIFF (accepted, en plain -> es plural) ${k}: en[${a}] es[${b}]`); paramMismatch++; }
}
say(`messages parsed: ${enKeys.length}/${enKeys.length} | parse failures: ${parseFail}`);

function placeholderSet(params) {
	return [...params.keys()].sort().join(",");
}
function paramKindCheck() {
	let n = 0;
	for (const k of enKeys) {
		const pe = M.parseMessage(en[k]), ps = M.parseMessage(es[k]);
		const a = [...pe.params.keys()].sort().join(","), b = [...ps.params.keys()].sort().join(",");
		if (a !== b) n++;
	}
	return n;
}
function paramKindCount() { return paramKindCheck(); }
say(`placeholder NAME sets identical: ${paramKindCheck() === 0 ? "yes (38/38)" : `${paramKindCheck()} mismatches`}`);
say(`placeholder KIND differences (en plain -> es plural): ${kindDiffs().map((k) => k.replace("mobile.schedules.", "")).join(", ") || "(none)"}`);
function kindDiffs() {
	const out = [];
	for (const k of enKeys) {
		const pe = M.parseMessage(en[k]), ps = M.parseMessage(es[k]);
		for (const [name, kind] of pe.params) {
			if (ps.params.get(name) !== kind) out.push(`${k}:${name}(${kind}->${ps.params.get(name)})`);
		}
	}
	return out;
}

H("3. PLURAL ARM STRUCTURE (es) — ALL CLDR CATEGORIES FOR es PRESENT, NO ARM DROPPED");
const cldr = new Intl.PluralRules("es").resolvedOptions().pluralCategories;
say("Intl.PluralRules('es').resolvedOptions().pluralCategories =", JSON.stringify(cldr));
say("app pluralCategory('es', n):",
	[0, 1, 2, 1000000, 1500000].map((n) => `${n}=${M.pluralCategory("es", n)}`).join(" "));
let armProblems = 0, pluralMessages = 0, droppedArms = 0;
for (const k of enKeys) {
	const ps = M.parseMessage(es[k]);
	const pe = M.parseMessage(en[k]);
	const esArms = armsOf(ps.nodes, null), enArms = armsOf(pe.nodes, null);
	for (let i = 0; i < esArms.length; i++) {
		pluralMessages++;
		const p = esArms[i];
		const need = ["one", "many", "other"];
		const miss = need.filter((c) => !p.arms.includes(c));
		const bogus = p.arms.filter((c) => !cldr.includes(c) && !/^=\d+$/.test(c));
		/* "no arm dropped": every en arm must still exist in es */
		const enSet = enArms[i]?.arms ?? [];
		const dropped = enSet.filter((c) => !p.arms.includes(c));
		if (dropped.length) droppedArms++;
		const ok = miss.length === 0 && bogus.length === 0 && dropped.length === 0;
		if (!ok) armProblems++;
		say(`${(ok ? "ok" : "PROBLEM").padEnd(7)} ${k.replace("mobile.schedules.", "")} [${p.name}] arms=${JSON.stringify(p.arms)}` +
			(miss.length ? ` MISSING=${JSON.stringify(miss)}` : "") +
			(bogus.length ? ` NOT-ES-CATEGORY=${JSON.stringify(bogus)}` : "") +
			(dropped.length ? ` DROPPED-vs-en=${JSON.stringify(dropped)}` : ""));
	}
}
say(`plural messages in es: ${pluralMessages} | with an arm problem: ${armProblems} | arms dropped vs en: ${droppedArms}`);
